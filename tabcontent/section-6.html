Demo 1 - Using filter name (built-in or custom)

Code

// HTML
<dropdowntree class="col-xs-5" 
	  ng-model="selectedItems6" 
	  data-settings="settings6"></dropdowntree>
	  
// Javascript

app.filter("myItemFilter", function(){
		return function(value, item, level){
				return item.text + " - " + item.value;
		};
	});
	
$scope.settings6 = {
  nameMember: "text",
  childrenMember: "items",
  sourceData: data,
  itemFormatter: "myItemFilter",
  templateUrl: templateUrl
};

Demo 2 - Using function

Code

// HTML
<dropdowntree class="col-xs-5" 
	  ng-model="selectedItems6" 
	  data-settings="settings7"></dropdowntree>
	  
// Javascript
$scope.settings6 = {
  nameMember: "text",
  childrenMember: "items",
  sourceData: data,
  itemFormatter: function(value, item, level){
	if(item.items && item.items.length > 0)
	{
		return value + " (<mark>" + item.items.length + "</mark>)";
	}
	
	return value;
  },
  templateUrl: templateUrl
};
